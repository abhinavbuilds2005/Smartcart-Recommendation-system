pd.read_csv('smartcart_customers.csv')\nprint(f\"Dataset shape: {df.shape}\")  # (2200, 29)\nprint(f\"Missing values:\\n{df.isnull().sum()}\")\nprint(f\"Data types:\\n{df.dtypes}\")\n```\n\n### Phase 2: Feature Engineering\n```python\n# Calculate Age\ndf['Age'] = pd.Timestamp.now().year - df['Birth_Year']\n\n# Calculate Customer Tenure\ndf['Customer_Tenure_Days'] = (df['Last_Purchase_Date'] - df['Enrollment_Date']).dt.days\n\n# Aggregate Spending\ndf['Total_Spending'] = (df['MntWines'] + df['MntFruits'] + \n                        df['MntMeats'] + df['MntFish'] + \n                        df['MntSweets'] + df['MntGold'])\n\n# One-Hot Encode Categoricals\ndf_encoded = pd.get_dummies(df[['Education', 'Marital_Status']], drop_first=True)\n```\n\n### Phase 3: Feature Scaling & Normalization\n```python\nfrom sklearn.preprocessing import StandardScaler\n\nscaler = StandardScaler()\nX_scaled = scaler.fit_transform(df[feature_cols])\n```\n\n### Phase 4: Dimensionality Reduction (PCA)\n```python\nfrom sklearn.decomposition import PCA\n\npca = PCA(n_components=3)\nX_pca = pca.fit_transform(X_scaled)\n\nprint(f\"Explained variance ratio: {pca.explained_variance_ratio_}\")  # [0.35, 0.28, 0.15]\nprint(f\"Total variance retained: {sum(pca.explained_variance_ratio_):.2%}\")  # ~95%\n```\n\n### Phase 5: Optimal Cluster Detection\n```python\nfrom sklearn.cluster import KMeans\nfrom sklearn.metrics import silhouette_score\n\n# Elbow Method\ninertias = []\nsilhouette_scores = []\n\nfor k in range(2, 11):\n    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)\n    labels = kmeans.fit_predict(X_pca)\n    inertias.append(kmeans.inertia_)\n    silhouette_scores.append(silhouette_score(X_pca, labels))\n\n# Optimal k = 4 (elbow point + highest silhouette score)\n```\n\n### Phase 6: Customer Segmentation\n```python\nkmeans_final = KMeans(n_clusters=4, random_state=42, n_init=10)\nlabels = kmeans_final.fit_predict(X_pca)\n\ndf['Cluster'] = labels\n\n# Segment Summary\ncluster_summary = df.groupby('Cluster')[feature_cols].mean()\nprint(cluster_summary)\n```\n\n### Phase 7: Churn Prediction Model\n```python\nfrom sklearn.ensemble import RandomForestClassifier\nfrom sklearn.model_selection import train_test_split\nfrom sklearn.metrics import classification_report, roc_auc_score\n\n# Prepare data\nX_churn = df[churn_features]\ny_churn = df['Churn']\n\nX_train, X_test, y_train, y_test = train_test_split(\n    X_churn, y_churn, test_size=0.2, random_state=42, stratify=y_churn\n)\n\n# Train model\nchurn_model = RandomForestClassifier(\n    n_estimators=100,\n    max_depth=10,\n    min_samples_split=5,\n    min_samples_leaf=2,\n    random_state=42,\n    n_jobs=-1\n)\nchurn_model.fit(X_train, y_train)\n\n# Evaluate\ny_pred = churn_model.predict(X_test)\ny_pred_proba = churn_model.predict_proba(X_test)[:, 1]\n\nprint(f\"Accuracy: {(y_pred == y_test).mean():.2%}\")\nprint(f\"ROC-AUC: {roc_auc_score(y_test, y_pred_proba):.4f}\")\nprint(f\"\\nClassification Report:\\n{classification_report(y_test, y_pred)}\")\n```\n\n---\n\n## 📊 Model Performance Metrics\n\n### Segmentation Model\n|
